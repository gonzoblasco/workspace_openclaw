**2. Palíndromo**
```js
function isPalindrome(str) {
  const clean = str.toLowerCase().replace(/[^a-z0-9]/g, "");
  return clean === clean.split("").reverse().join("");
}
```

**3. Contar vocales**
```js
function countVowels(str) {
  const counts = { a: 0, e: 0, i: 0, o: 0, u: 0 };
  for (const ch of str.toLowerCase()) {
    if (ch in counts) counts[ch]++;
  }
  return counts;
}
```

**4. Factorial iterativo**
```js
function factorial(n) {
  let result = 1;
  for (let i = 2; i <= n; i++) result *= i;
  return result;
}
```

**5. Pares de un array**

function evens(arr) {

**6. Invertir string**
```js
function reverse(str) {
  let result = "";
  for (let i = str.length - 1; i >= 0; i--) result += str[i];
  return result;
}
```

**7. Máximo y mínimo**
```js
function minMax(arr) {
  let min = arr[0], max = arr[0];
  for (const n of arr) {
    if (n < min) min = n;
    if (n > max) max = n;
  }
  return { min, max };
}
```

**8. Suma de dígitos**
```js
function sumDigits(n) {
  return String(n)
    .split("")
    .reduce((acc, d) => acc + Number(d), 0);
}
```

**9. Anagrama**
```js
function isAnagram(a, b) {
  const norm = (s) => s.toLowerCase().split("").sort().join("");
  return norm(a) === norm(b);
}
```

**10. Fibonacci**
```js
function fibonacci(n) {
  if (n <= 0) return [];
  if (n === 1) return [0];
  const seq = [0, 1];
  for (let i = 2; i < n; i++) {
    seq.push(seq[i - 1] + seq[i - 2]);
  }
  return seq;
}
```
